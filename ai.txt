-- // 4. Raycast Ground Checker \\ --
local function isGroundSafe(targetPos)
    local raycastParams = RaycastParams.new()
    raycastParams.FilterDescendantsInstances = {LocalPlayer.Character}
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    local raycastResult = Workspace:Raycast(targetPos + Vector3.new(0, 2, 0), Vector3.new(0, -15, 0), raycastParams)
    return raycastResult ~= nil
end

-- // 5. Advanced Engine (GodMode, Anti-Void, Anti-Fling, FullBright) \\ --
local lastSafeGroundCFrame = nil

RunService.Stepped:Connect(function()
    local hrp = getHRP()
    local hum = getHumanoid()
    
    if hrp then
        if hrp.Position.Y > -5 and not _G.SafeAntiGlitchActive then
            if isGroundSafe(hrp.Position) then
                lastSafeGroundCFrame = hrp.CFrame
            end
        end

        if hrp.Position.Y < -20 then
            if lastSafeGroundCFrame then
                hrp.CFrame = lastSafeGroundCFrame + Vector3.new(0, 3, 0)
            else
                hrp.CFrame = CFrame.new(0, 10, 0)
            end
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end
        
        if antiFlingEnabled then
            hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        end
    end

    if godModeEnabled and hum then
        hum.Health = hum.MaxHealth
        for _, part in pairs(LocalPlayer.Character:GetDescendants()) do
            if part:IsA("TouchTransmitter") then
                pcall(function() part:Destroy() end)
            end
        end
    end

    if noclipEnabled and LocalPlayer.Character then
        for _, part in pairs(LocalPlayer.Character:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end

    if hum and hum.WalkSpeed ~= currentWalkSpeed then
        hum.WalkSpeed = currentWalkSpeed
    end

    if fullBrightEnabled then
        Lighting.Brightness = 2
        Lighting.ClockTime = 14
        Lighting.GlobalShadows = false
    end
end)

-- // 6. Smart Role Finders \\ --
local function getMurdererPlayer()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local vChar = p.Character
            local backpack = p:FindFirstChild("Backpack")
            local hasKnife = vChar:FindFirstChild("Knife") or (backpack and backpack:FindFirstChild("Knife"))
            if hasKnife then
                return p
            end
        end
    end
    return nil
end

local function getSheriffPlayer()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local char = p.Character
            local bp = p:FindFirstChild("Backpack")
            if char:FindFirstChild("Gun") or (bp and bp:FindFirstChild("Gun")) then
                return p
            end
        end
    end
    return nil
end

local function isLocalMurderer()
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    return (char and char:FindFirstChild("Knife")) or (backpack and backpack:FindFirstChild("Knife"))
end

-- // 8. Sheriff Combat Aimbot \\ --
RunService.RenderStepped:Connect(function()
    local hrp = getHRP()
    if not hrp or isLocalMurderer() then return end
    
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    
    local equippedGun = char and char:FindFirstChild("Gun")
    local backpackGun = backpack and backpack:FindFirstChild("Gun")
    local hasGunInPossession = (equippedGun ~= nil or backpackGun ~= nil)

    local murderer = getMurdererPlayer()
    
    if hasGunInPossession and murderer and murderer.Character and combatProEnabled then
        local mHRP = murderer.Character:FindFirstChild("HumanoidRootPart")
        local mHum = murderer.Character:FindFirstChildOfClass("Humanoid")
        
        if mHRP and mHum and mHum.Health > 0 then
            local predictedPos = mHRP.Position + (mHRP.AssemblyLinearVelocity * 0.05)
            local dist = (hrp.Position - predictedPos).Magnitude
            
            local targetCF
            if dist < 14 then
                local strafeOffset = Vector3.new(math.sin(tick() * 15) * 8, 2, math.cos(tick() * 15) * 8)
                targetCF = CFrame.new(predictedPos + strafeOffset, predictedPos)
            else
                targetCF = CFrame.new(predictedPos + Vector3.new(0, 0, 12), predictedPos)
            end
            
            hrp.CFrame = hrp.CFrame:Lerp(targetCF, 0.7)
            Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, predictedPos + Vector3.new(0, 1.0, 0))
            
            if autoShootEnabled then
                local hum = getHumanoid()
                if backpackGun and not equippedGun and hum then
                    hum:EquipTool(backpackGun)
                end
                local gun = char and char:FindFirstChild("Gun")
                if gun then
                    pcall(function()
                        gun:Activate()
                        local cx, cy = Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2
                        VirtualInputManager:SendMouseButtonEvent(cx, cy, 0, true, game, 0)
                        task.wait()
                        VirtualInputManager:SendMouseButtonEvent(cx, cy, 0, false, game, 0)
                    end)
                end
            end
        end
    end
end)

local function getGunDrop()
    for _, obj in pairs(Workspace:GetChildren()) do
        if obj.Name == "GunDrop" or obj:FindFirstChild("GunVisual") or (obj:IsA("Tool") and obj.Name == "Gun") then
            return obj
        end

local sheriffHasDied = false
local lastGunPos = nil
local grabInProgress = false

RunService.Heartbeat:Connect(function()
    local sheriff = getSheriffPlayer()
    if sheriff and sheriff.Character then
        local sHum = sheriff.Character:FindFirstChildOfClass("Humanoid")
        local sHRP = sheriff.Character:FindFirstChild("HumanoidRootPart")
        if sHum and sHRP then
            if sHum.Health > 0 then
                sheriffHasDied = false
                lastGunPos = sHRP.CFrame
            else
                sheriffHasDied = true
            end
        end
    else
        local drop = getGunDrop()
        if drop then
            local part = drop:IsA("Model") and drop.PrimaryPart or drop
            if part and part:IsA("BasePart") then
                lastGunPos = part.CFrame
                sheriffHasDied = true
            end
        end
    end
end)

task.spawn(function()
    while true do
        task.wait(0.1)
        if autoGrabGunEnabled and not grabInProgress and not isLocalMurderer() then
            local charCheck = LocalPlayer.Character
            local bpCheck = LocalPlayer:FindFirstChild("Backpack")
            local hasGunAlready = (charCheck and charCheck:FindFirstChild("Gun")) or (bpCheck and bpCheck:FindFirstChild("Gun"))
            
            if sheriffHasDied and lastGunPos and not hasGunAlready then
                local hrp = getHRP()
                if hrp and isGroundSafe(lastGunPos.Position) then
                    grabInProgress = true
                    local originalPos = hrp.CFrame
                    
                    _G.SafeAntiGlitchActive = true
                    hrp.CFrame = lastGunPos + Vector3.new(0, 1.5, 0)
                    task.wait(0.15) 
                    
                    hrp.CFrame = originalPos
                    
                    task.delay(0.1, function()
                        _G.SafeAntiGlitchActive = false
                    end)
                    
                    sheriffHasDied = false
                    task.wait(1.5)
                    grabInProgress = false
                end
            end
        end
    end
end)

local function removeESP(player)
    if espCache[player] then
        if espCache[player].highlight then espCache[player].highlight:Destroy() end
        if espCache[player].tracer then espCache[player].tracer:Remove() end
        if espCache[player].textTag then espCache[player].textTag:Remove() end
        espCache[player] = nil
    end
end

local function setupESP(player)
    if player == LocalPlayer then return end
    removeESP(player)

    local highlight = Instance.new("Highlight")
    highlight.Name = "LuxHighlight"
    highlight.FillTransparency = 0.35
    highlight.OutlineTransparency = 0.05

    local tracer = Drawing.new("Line")
    tracer.Visible = false
    tracer.Thickness = 1.5

    local textTag = Drawing.new("Text")
    textTag.Visible = false
    textTag.Size = 12
    textTag.Center = true
    textTag.Outline = true

    espCache[player] = {highlight = highlight, tracer = tracer, textTag = textTag}
end

Players.PlayerRemoving:Connect(removeESP)
for _, p in pairs(Players:GetPlayers()) do setupESP(p) end
Players.PlayerAdded:Connect(setupESP)

RunService.RenderStepped:Connect(function()
    for player, cache in pairs(espCache) do
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local backpack = player:FindFirstChild("Backpack")

        if char and hrp and hitboxEnabled and player ~= LocalPlayer then
            hrp.Size = Vector3.new(4, 4, 4)
            hrp.Transparency = 0.7
        elseif char and hrp and not hitboxEnabled and player ~= LocalPlayer then
            hrp.Size = Vector3.new(2, 2, 1)
            hrp.Transparency = 1
        end

        if espEnabled and char and hrp and cache then
            local roleColor = Color3.fromRGB(0, 255, 127)
            local roleName = "Innocent"

            local hasKnife = char:FindFirstChild("Knife") or (backpack and backpack:FindFirstChild("Knife"))
            local hasGun = char:FindFirstChild("Gun") or (backpack and backpack:FindFirstChild("Gun"))

            if hasKnife then
                roleColor = Color3.fromRGB(255, 40, 40)
                roleName = "Murderer"
            elseif hasGun then
                roleColor = Color3.fromRGB(0, 162, 255)
                roleName = "Sheriff"
            end

            cache.highlight.Parent = char
            cache.highlight.FillColor = roleColor
            cache.highlight.OutlineColor = roleColor

            local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
            local myHRP = getHRP()

            if onScreen and myHRP then
                local distance = math.floor((myHRP.Position - hrp.Position).Magnitude)
                
                cache.tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                cache.tracer.To = Vector2.new(screenPos.X, screenPos.Y)
                cache.tracer.Color = roleColor
                cache.tracer.Visible = true

                cache.textTag.Position = Vector2.new(screenPos.X, screenPos.Y - 25)
                cache.textTag.Text = string.format("%s (%s) | [%d m]", player.Name, roleName, distance)
                cache.textTag.Color = roleColor
                cache.textTag.Visible = true
            else
                cache.tracer.Visible = false
                cache.textTag.Visible = false
            end
        else
            if cache.highlight then cache.highlight.Parent = nil end
            if cache.tracer then cache.tracer.Visible = false end
            if cache.textTag then cache.textTag.Visible = false end
        end
    end
end)

-- // 11. Coin Collector \\ --
local function findActiveCoinContainer()
    for _, obj in pairs(Workspace:GetChildren()) do
        if obj:IsA("Model") and obj:FindFirstChild("CoinContainer") then
            return obj.CoinContainer
        end
    end
    local direct = Workspace:FindFirstChild("CoinContainer")
    if direct then return direct end
    return nil
end

local function findNearestCoin(coinContainer)
    local nearestCoin = nil
    local shortestDistance = math.huge
    local hrp = getHRP()

    if coinContainer and hrp then
        for _, coin in pairs(coinContainer:GetChildren()) do
            if coin:IsA("BasePart") and not visitedCoins[coin] and coin:FindFirstChild("CoinVisual") then
                local distance = (hrp.Position - coin.Position).Magnitude
                if distance < shortestDistance then
                    shortestDistance = distance
                    nearestCoin = coin
                end
            end
        end
    end
    return nearestCoin
end

task.spawn(function()
    while true do
        task.wait(COIN_DELAY) 
        if isRunning and not isLocalMurderer() then
            local hrp = getHRP()
            local coinContainer = findActiveCoinContainer()
            
            if hrp and coinContainer then
                local targetCoin = findNearestCoin(coinContainer)
                if targetCoin and isGroundSafe(targetCoin.Position) then
                    visitedCoins[targetCoin] = true
                    hrp.CFrame = targetCoin.CFrame + Vector3.new(0, 0.5, 0)
                end
            end
        end
    end
end)

-- // 12. Role Teleports \\ --
local function teleportToRole(itemName)
    for _, target in pairs(Players:GetPlayers()) do
        if target ~= LocalPlayer and target.Character then
            local char = target.Character
            local backpack = target:FindFirstChild("Backpack")
            if (char and char:FindFirstChild(itemName)) or (backpack and backpack:FindFirstChild(itemName)) then
                local myHRP = getHRP()
                local targetHRP = char:FindFirstChild("HumanoidRootPart")
                if myHRP and targetHRP then
                    myHRP.CFrame = targetHRP.CFrame * CFrame.new(0, 3, 0)
                end
                break
            end
        end
    end
end

    isMinimized = not isMinimized
    if isMinimized then
        TabBar.Visible = false
        PagesContainer.Visible = false
        MainFrame:TweenSize(UDim2.new(0, 380, 0, 30), Enum.EasingDirection.Out, Enum.EasingStyle.Quart, 0.15, true)
        MinimizeBtn.Text = "+"
    else
        MainFrame:TweenSize(UDim2.new(0, 380, 0, 250), Enum.EasingDirection.Out, Enum.EasingStyle.Quart, 0.15, true)

        PagesContainer.Visible = true
        MinimizeBtn.Text = "-"
    end
end)
